\section*{A.1 关键符号}
\begin{itemize}
  \item $\mathbf s(\alpha,\beta)$：目标方向单位向量；
  \item $\mathbf x_i,\mathbf x_i'$：节点调节前后坐标；
  \item $\Delta l_i$：第 $i$ 个促动器径向伸缩量；
  \item $\eta$：馈源舱接收比。
\end{itemize}

\section*{A.2 关键约束}
\begin{itemize}
  \item 促动器行程约束：$[-0.6,0.6]$ m；
  \item 相邻节点距离相对变化不超过 $0.07\%$。
\end{itemize}
