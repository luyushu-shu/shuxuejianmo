\chapter{排版建议}

这里介绍一些用 \LaTeX{} 书写论文时的注意事项和排版建议, 仅供参考.

\section{源文件的书写}

\begin{itemize}
  \item \myemph{整篇论文的标点符号要统一}.
  \begin{enumerate}[label=$\triangleright$]
    \item 数学公式和英文摘要中使用英文标点符号;
    \item 中文部分可以用中文标点，或者英文标点, 但要统一;
    \item 其他符号, 如小括号 \Verb|()|, 问号 \Verb|?| 等, 也要与中文部分的标点符号统一.
  \end{enumerate}

  \item \myemph{英文标点后面要留空格}.

  \item \myemph{中文和英文之间, 中文与数学公式 (包括单个数学变量) 之间,
   中文与参考文献、定理引理、算法、公式等的引用之间 $\to$ 用空格隔开}.

  \item 左双引号 $\to$ 两个连续的 ``反引号" (通常位于键盘左上角).

  \item \Verb|\ldots| 和 \Verb|\cdots| 的使用场合, 例如:
    $k=1,2,\ldots,n$,\quad $f=1+2+\cdots+n$.

  \item 页面不要出现大片空白 (页面存在浮动表格或图片时, 可能会现大段空白, 此时可以做适当调整).

  \item \myemph{参考文献格式要统一}: 相关建议详见第 \ref{sec:Advice-bib} 节.
  % 作者, 标题, 杂志名称, 年份, 卷期, 页码
%
%  \item 参考文献的标签命名, 多名作者: 姓 (前三个作者) + 年; 单个作者: 姓 (前三个字母) + 年

  \item \myemph{没有被引用的公式无需编号, 没有被引用的文献不用出现在论文末尾的参考文献列表中}.

  \item \myemph{数学函数}一律用直立字体, 可以通过 \Verb|\DeclareMathOperator| 定义, 如: \\
     \Verb|\DeclareMathOperator{\rank}{rank}|, \\
     \Verb|\DeclareMathOperator{\diag}{diag}|.

     如果只是临时使用, 也可以用 \Verb|\operatorname| 命令, 如

     \Verb|$\operatorname{span}\{v_1,v_2\}$|

\end{itemize}

\bigskip
\begin{Block}[常见排版问题]
\begin{itemize}
  \item 遗漏行间数学公式后面的标点符号.
  \item 数学函数名, 如 $\sin,\log,\max$ 等, 没有用直立字体 (函数对应的命令).
  \item 中文与参考文献的引用之间, 中文与数学公式的引用之间, 没有留空格.
  \item 左双引号.
\end{itemize}
\end{Block}

\section{源文件书写建议}

%以下是对书写 \Verb|tex| 源文件的一些建议, 供参考.
\begin{itemize}
  \item 各种环境的开始 \Verb|\begin{...}|
     和结束 \Verb|\end{...}| $\to$ 独占一行.

  \item 自然分段 $\to$ 建议使用一个空行, 增加源文件可读性.

  \item 自然分行, 慎用强制换行符 ``\Verb|\\|".
  \item[] 如果由于内容超越了右边界而只能通过手工换行时,
   建议使用 \Verb|\linebreak|, 可以避免出现右边空白
   (类似 Word 中两边对齐效果)

  \item \myemph{源文件一行不要太长, 特别地, 不要一整段都写在一行上}, 建议在适当的地方换行,
    增强源文件的可读性, 也便于查找错误.

  \item \myemph{如果数学公式太长 (源代码, 不是指公式本身), 建议在适当的地方换行}.

  \item 多行数学公式: 建议使用 \Verb|align| (带编号) 或 \Verb|align*| (不带编号).
    另外, 建议源文件也尽量对齐.
    \begin{align}
      f(x,y) & = (x+y)^2 \label{eq:fxy-align-a} \\
             & = x^2 + 2xy + y^2. \label{eq:fxy-align-b}
    \end{align}
    源代码如下所示:

%            & = x^2 + 2xy + y^2. \label{eq:fxy-align-b}
%  \end{align}
%\end{syntax}

  \begin{center}
    \boxed{\includegraphics[width=0.75\linewidth]{fig_align.png}}
  \end{center}

  \item 多行数学公式: 每行都居中对齐, 可用 \Verb|gather| 或 \Verb|gather*| 环境.
    \begin{gather}
      f(x,y) = (x+y)^2 \\
      g(x,y) = x^2 + 2xy + y^2.
    \end{gather}

  \item 多行数学公式: 整体一个编号, 可以使用 \Verb|align + \notag| 
    \begin{align}
      f(x,y) & = (x+y)^2 \notag \\
             & = x^2 + 2xy + y^2. \label{eq:fxy-align-c}
    \end{align}

    \boxed{\includegraphics[width=0.75\linewidth]{fig_align_notag.png}}
  \end{center}
    
    
    或 \Verb|equation + aligned| (公式编号垂向位置居中)
    \begin{equation} \label{eq:fxy-aligned}
      \begin{aligned}
        f(x,y) & = (x+y)^2 \\
               & = x^2 + 2xy + y^2.
      \end{aligned}
    \end{equation} 
    
  \begin{center}
    \boxed{\includegraphics[width=0.75\linewidth]{fig_equation_aligned.png}}
  \end{center}

\begin{Alert}
  多行数学公式不建议用 \Verb|eqnarray|.
\end{Alert}

\item 数学公式的引用: 建议使用 \Verb|\eqref{...}|,
如 \Verb|\eqref{eq:fxy-align-b}|, 输出结果为 \eqref{eq:fxy-align-b}.

\item 数学公式手工编号: \Verb|equation*| 环境 + \Verb|\tag| 或 \Verb|\tag*|,
  可正常引用, 如 \eqref{eq:taga}
    \begin{equation*}
      f(x,y)  = (x+y)^2 \label{eq:taga} \tag{A}
    \end{equation*}

  \begin{center}
    \boxed{\includegraphics[width=0.75\linewidth]{fig_equation_tag.png}}
  \end{center}
\end{itemize}


\section{参考文献书写建议}
\label{sec:Advice-bib}

\begin{itemize}
  \item \myemph{排序}: 建议按作者姓氏的字母排序.

  \item 参考文献的标签命名 (也可以与 zotero 中的 Better BibTeX 保持一致)
  \begin{itemize}
    \item 单个作者, 采用 ``姓 (前三个字母) + 年份 (后两位)", 如 ``\Verb|Hua16|"
    \item 2-3 个作者, 采用 ``姓 (首字母) + 年份", 如 ``\Verb|DDH07|"
    \item 超过 3 个作者, 采用 ``前三个作者的姓 (首字母) + 加号 + 年份", 如 ``\Verb|XTZ+18|"
  \end{itemize}
  \begin{center}
    \boxed{\includegraphics[width=0.85\linewidth]{bib_art.png}}
  \end{center}


  \item 格式要求: 普通论文 (示例源文件见上图)
  \begin{itemize}
    \item 作者名: 格式统一, 可以是 ``名" 在前, ``姓" 在后, 如 ``\Verb|J. Demmel|";
      作者之间用分号隔开; 如果是多个作者, 最后一个作者前加 ``and"; 如果存在 Middle name,
      则 First name 和 Middle name 的缩写之间可以加空格, 如 ``\Verb|J. Y. Huang|".
    \item 标题: 除第一个单词的首字母和专有名词外, 一律小写.
    \item 杂志名: 首字母大写; 可以缩写, 也可以全名, 但这篇论文要统一; 建议加 \Verb|\emph{...}|
    \item 卷, 年份, 页码: ``卷 (年份), 页码范围", 其中页码范围用两个连续的英文减号, 如 ``\Verb|106 (2007), 199--224|"
    \item 其他: DOI, 网址等, 这些可选.
  \end{itemize}


  \item 格式要求: 图书 (含博士/硕士论文, 讲义, 示例源文件见下图)
  \begin{itemize}
    \item 作者名: 同论文.
    \item 标题: 首字母大写.
    \item 出版社, 城市, 年份.
  \end{itemize}
  \begin{center}
    \boxed{\includegraphics[width=0.95\linewidth]{bib_book.png}}
  \end{center}

\end{itemize}



\chapter{常见排版技巧}

\section{数学公式}

\begin{itemize}
  \item 矩阵 $\to$ \verb|amsmath| 宏包提供了矩阵环境, 如 \Verb|bmatrix| 或 \Verb|pmatrix|
  \item[] \Verb|\begin{bmatrix} A & B \\ B^\T & C \end{bmatrix}|
    $\to$ $\bbm A & B \\ B^\T & C \ebm$
  \item[] \Verb|\begin{pmatrix} 1 & 2 \\ -1 & 123 \end{pmatrix}|
    $\to$ $\begin{pmatrix} 1 & 2 \\ -1 & 123 \end{pmatrix}$
  \item[] (缺点: 列格式缺省是居中对齐, 无法修改, 解决方法: 用 \Verb|mathtools| 宏包)

%  \item \verb|mathtools| 宏包提供了带 ``*" 的矩阵环境, 可以修改列格式:

  \item 自定义命令, 如: \\
    \Verb|\newcommand{\bbm}{\begin{bmatrix}}| \\
    \Verb|\newcommand{\ebm}{\end{bmatrix}}|

  \item 自适应长度 (可变高度) 的边界符 $\to$ \Verb|\left|, \Verb|\right|, 如:
  \item[] \Verb<|\dfrac{a-1}{a+1}|<
    $\to$ $|\dfrac{a-1}{a+1}|$
  \item[] \Verb<\left|\dfrac{a-1}{a+1}\right|<
    $\to$ $\left|\dfrac{a-1}{a+1}\right|$

  \item 公式越界问题: 手工分割.
\end{itemize}

\bigskip
\begin{Block}[mathtools 宏包]
  对 \Verb|amsmath| 进行了扩展, 自动加载 \Verb|amsmath| 宏包, 提供以下功能 (部分)
  \begin{itemize}
    \item
      \Verb|\mathtoolsset{showonlyrefs=true}|  \\
      只对被引用的公式编号 (引用公式时需用 \Verb|eqref|)

    \item 可指定对齐方式的矩阵环境: 
      \Verb|matrix*, pmatrix*, bmatrix*, ...| \\
      如右对齐 \Verb|\begin{bmatrix*}[r] ... | \\
 {\small \Verb|\begin{bmatrix*}[r] 1 & -2 \\ -1 & 3 \end{bmatrix*}|
    $\to$ $\begin{bmatrix*}[r] 1 & -2 \\ -1 & 3 \end{bmatrix*}$ }

    \item 左角标: \Verb|\prescript{sup}{sub}{arg}| \\
    {\Verb|\prescript{\mathrm{RL}}{a}{\mathrm{D}}^{x}_{c}|}
    $\to\prescript{\mathrm{RL}}{a}{\mathrm{D}}^{x}_{c}$
  \end{itemize}
\end{Block}

\subsection{算法伪代码示例 --- algorithm 环境}

算法 \ref{Alg:MGS-QR} 是用 \Verb|algorithm| 环境排版的伪代码.
缺点是不支持换页, 可能会出现大片空白.

\begin{algorithm}[H]
\caption{基于 MGS 的 QR 分解\label{Alg:MGS-QR}}
\begin{algorithmic}[1]
  \Statex\hspace*{-2em}
     \Comment{Given $A$, compute $Q=[q_1,\ldots,q_n]$ and $R$ such that $A=QR$}
  \State Set $R=[r_{ij}]=0_{n\times n}$  (the $n\times n$ zero matrix)
  \If{$a_1=0$}
  \State $q_1=0$
  \Else
  \State $r_{11}=\|a_1\|_2$,\quad $q_1 = a_1/\|a_1\|_2$
  \EndIf
  \For{$j=2$ to $n$}
  \State $q_j = a_j$
  \For{$i=1$ to $j-1$} \Comment{MGS, 注意与 GS 的区别 }
  \State $r_{ij} = (q_j,q_i)$,\quad $q_j = q_j - r_{ij}q_i$
  \EndFor
  \If{$q_j\neq 0$}
  \State $r_{jj}=\|q_j\|_2$,\quad  $q_j = q_j/r_{jj}$
  \EndIf
  \EndFor
\end{algorithmic}
\end{algorithm}


\subsection{算法伪代码示例 --- breakablealgorithm 环境}

算法 \ref{Alg:GMRES} 是用 \Verb|breakablealgorithm| 环境排版的伪代码, 可以跨页显示.

\begin{breakablealgorithm}
\caption{实用 GMRES 方法\label{Alg:GMRES}}
\begin{algorithmic}[1]
  \State 选取初值 $x^{(0)}$, 停机标准 $\varepsilon>0$, 以及最大迭代步数 IterMax
  \State $r_0=b-Ax^{(0)}$, $\beta=\|r_0\|_2$
  \If{$\beta/\|b\|_2<\varepsilon$}
  \State 停止计算, 输出近似解 $x^{(0)}$
  \EndIf
  \State $v_1=r_0/\beta$
  \State $\xi=\beta e_1$ \Comment{记录 $q_1$}
  \For{$j = 1$ to IterMax}
  \State $w = Av_j$
  \For{$i=1$ to $j$} \Comment{Arnoldi 过程}
  \State $h_{i,j} = (v_i,w)$
  \State $w = w - h_{i,j}v_i$
  \EndFor
  \State $h_{j+1,j}=\|w\|_2$
  \If{$h_{j+1,j}=0$} \Comment{迭代中断}
  \State $m=j$, break
  \EndIf
  \State $v_{j+1}=w/h_{j+1,j}$
  \For{$i=1$ to $j-1$}
         \Comment{计算 $G_{j-1}\cdots G_2G_1 H_{j+1,j}(1\!:\!j,j)$}
  \smallskip
  \State $\bbm h_{ij}\\ h_{i+1,j} \ebm = \bbm c_i & s_i \\ -s_i & c_i \ebm
         \bbm h_{ij}\\ h_{i+1,j} \ebm $
  \smallskip
  \EndFor
  \If{$|h_{jj}|>|h_{j+1,j}|$} \Comment{构造 Givens 变换 $G_j$}
  \State $\tau=h_{j+1,j}/h_{jj}$, $c_j=1/\sqrt{1+\tau^2}$, $s_j=c_j\tau$
  \Else
  \State $\tau=h_{jj} / h_{j+1,j}$, $s_j=1/\sqrt{1+\tau^2}$, $c_j=s_j\tau$
  \EndIf
  \State $h_{jj}=c_jh_{jj}+s_jh_{j+1,j}$
        \Comment{计算 $G_{j} H_{j+1,j}(1\!:\! j,j)$}
  \State $h_{j+1,j}=0$ \smallskip
  \State $\bbm \xi_j \\ \xi_{j+1} \ebm = \bbm c_j & s_j \\ -s_j & c_j \ebm
         \bbm \xi_{j}\\ 0 \ebm $
        \Comment{计算 $G_{j} (\beta G_{j-1}\cdots G_2G_1 e_1)$}
  \smallskip
  \State relres = $|\xi_{j+1}|/\beta$ \Comment{相对残量}
  \If{$\mathrm{relres}<\varepsilon$} % \Comment{收敛检测}
  \State $m=j$, break
  \EndIf
  \EndFor
  \State $m=j$
  \State $y^{(m)}=H(1:m,1:m) \backslash \xi(1:m)$
        \Comment{求最小二乘问题的解, 回代求解}
  \State $x^{(m)}=x^{(0)}+V_m y^{(m)}$
  \If{$\mathrm{relres}<\varepsilon$}
  \State 输出近似解 $x$ 及相关信息
  \Else
  \State 输出算法失败信息
  \EndIf
\end{algorithmic}
\end{breakablealgorithm}
